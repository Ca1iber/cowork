@tilelang.jit(
    out_idx=[],
    pass_configs={
        tilelang.PassConfigKey.TL_ENABLE_FAST_MATH: True,
        tilelang.PassConfigKey.TL_DISABLE_WARP_SPECIALIZED: True,
    },
)
def _make_s8_score_pool(
    batch, seq_len, kv_heads, query_heads, dim, selected_blocks, block_size, is_causal,
):
    groups = query_heads // kv_heads
    assert selected_blocks == 8 and block_size == 16 and dim == 64 and groups == 16
    tokens = selected_blocks * block_size
    scale = (1.0 / dim) ** 0.5 * 1.44269504
    q_shape = [batch, seq_len, query_heads, dim]
    kv_shape = [batch, seq_len, kv_heads, dim]
    index_shape = [batch, seq_len, kv_heads, selected_blocks]
    dtype = T.float16
    accum_dtype = T.float32
    pool_layout = T.Fragment(
        [groups, tokens],
        forward_thread_fn=lambda row, col: row + 16 * ((col % 16) // 4),
        forward_index_fn=lambda row, col: (col // 16) * 4 + col % 4,
    )
    output_layout = T.Fragment(
        [groups, dim],
        forward_thread_fn=lambda row, col: row + 16 * ((col % 16) // 4),
        forward_index_fn=lambda row, col: (col // 16) * 4 + col % 4,
    )

    @T.prim_func
    def native_sparse_attention(
        Q: T.Tensor(q_shape, dtype),
        K: T.Tensor(kv_shape, dtype),
        V: T.Tensor(kv_shape, dtype),
        BlockIndices: T.Tensor(index_shape, T.int32),
        Output: T.Tensor(q_shape, dtype),
    ):
        with T.Kernel(seq_len, batch * kv_heads, threads=64) as (token, batch_head):
            batch_id = batch_head // kv_heads
            kv_head = batch_head % kv_heads
            lane_id = T.KernelLaunchFrame.Current().get_thread_binding() % 64
            q_local = T.alloc_local(16, dtype)
            k_local = T.alloc_local(4, dtype)
            v_local = T.alloc_local(16, dtype)
            scores = T.alloc_fragment([groups, tokens], accum_dtype)
            scores_half = T.alloc_fragment([groups, tokens], dtype)
            maximum = T.alloc_fragment([groups], accum_dtype)
            denominator = T.alloc_fragment([groups], accum_dtype)
            output_acc = T.alloc_fragment([groups, dim], accum_dtype)
            v_shared = T.alloc_shared([block_size, dim], dtype)
            output_shared = T.alloc_shared([groups, dim], dtype)
            T.annotate_layout({
                scores: pool_layout,
                scores_half: pool_layout,
                output_acc: output_layout,
                v_shared: T.Layout(
                    [block_size, dim],
                    lambda row, col: (row, ((col // 8) ^ (2 * (row // 4))) * 8 + col % 8),
                ),
            })
            for chunk in T.serial(4):
                for element in T.vectorized(4):
                    head = lane_id % 16
                    feature = chunk * 16 + (lane_id // 16) * 4 + element
                    q_local[chunk * 4 + element] = Q[
                        batch_id, token, kv_head * groups + head, feature
                    ]
            T.fill(scores, 0)
            for selected in T.unroll(selected_blocks):
                block_start = BlockIndices[batch_id, token, kv_head, selected] * block_size
                if block_start >= 0 and block_start <= token:
                    for chunk in T.serial(4):
                        for element in T.vectorized(4):
                            row = lane_id % 16
                            feature = chunk * 16 + (lane_id // 16) * 4 + element
                            k_local[element] = K[
                                batch_id, block_start + row, kv_head, feature
                            ]
                        T.tvm_mfma(
                            '16x16x16f16', 'row', 'row',
                            'float16x4', 'float16x4', 'float32x4',
                            k_local.data, 0, q_local.data, chunk, scores.data, selected,
                            dtype='float32x4',
                        )
            for head, offset in T.Parallel(groups, tokens):
                block_start = BlockIndices[batch_id, token, kv_head, offset // block_size] * block_size
                if is_causal:
                    scores[head, offset] = T.if_then_else(
                        block_start >= 0 and block_start + offset % block_size <= token,
                        scores[head, offset], -T.infinity(accum_dtype),
                    )
                else:
                    scores[head, offset] = T.if_then_else(
                        block_start >= 0 and block_start <= token and block_start + offset % block_size < seq_len,
                        scores[head, offset], -T.infinity(accum_dtype),
                    )
            T.fill(maximum, -T.infinity(accum_dtype))
            T.reduce_max(scores, maximum, dim=1, clear=True)
            for head, offset in T.Parallel(groups, tokens):
                scores[head, offset] = T.exp2(scores[head, offset] * scale - maximum[head] * scale)
            T.reduce_sum(scores, denominator, dim=1)
            T.copy(scores, scores_half)
            T.fill(output_acc, 0)
            for selected in T.unroll(selected_blocks):
                block_start = BlockIndices[batch_id, token, kv_head, selected] * block_size
                if block_start >= 0 and block_start <= token:
                    T.copy(
                        V[batch_id, block_start:block_start + block_size, kv_head, :],
                        v_shared,
                    )
                    for tile in T.serial(4):
                        for element in T.vectorized(4):
                            row = (lane_id // 16) * 4 + element
                            feature = tile * 16 + lane_id % 16
                            v_local[tile * 4 + element] = v_shared[row, feature]
                    for tile in T.serial(4):
                        T.tvm_mfma(
                            '16x16x16f16', 'row', 'row',
                            'float16x4', 'float16x4', 'float32x4',
                            v_local.data, tile, scores_half.data, selected, output_acc.data, tile,
                            dtype='float32x4',
                        )
            for head, feature in T.Parallel(groups, dim):
                output_acc[head, feature] /= denominator[head]
            T.copy(output_acc, output_shared)
            T.copy(
                output_shared,
                Output[batch_id, token, kv_head * groups:(kv_head + 1) * groups, :],
            )

    return native_sparse_attention
